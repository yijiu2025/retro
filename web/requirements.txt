# RetroChimera 网页版 —— Python 依赖
# ------------------------------------------------------------
# 🔴 这不是 `pip install -r` 一把梭的清单 —— 请**按顺序**看下面三段。
#    直接 `pip install -r requirements.txt` 在 Windows 上大概率会因为
#    torch / torch-scatter 的索引源问题失败。
#
# 实测可行组合（2026-09-22，Windows + RTX 4070）：
#   Python 3.9.7 | torch 2.2.2+cu121 | numpy 1.26.4 | torch-geometric 2.5.2
#   rdkit 2023.9.6 | syntheseus 0.8.0 | retrochimera 1.3.0

# ============================================================
# 【第 1 步】PyTorch —— 必须用 pip 官方 wheel，**不要用 conda**
#   conda 的 pytorch 2.2.2 win-64 构建与 conda-forge 的新版 intel-openmp 冲突，
#   运行时报 fbgemm.dll 加载失败（WinError 182）。
#
#   GPU（cu121）：
#     pip install torch==2.2.2 torchvision==0.17.2 \
#       --index-url https://download.pytorch.org/whl/cu121
#   纯 CPU（无 GPU 的机器）：
#     pip install torch==2.2.2 torchvision==0.17.2 \
#       --index-url https://download.pytorch.org/whl/cpu
# ============================================================
torch==2.2.2
torchvision==0.17.2

# numpy 必须锁 1.x（torch 2.2.2 是按 numpy 1.x 编译的）
numpy==1.26.4

# ============================================================
# 【第 2 步】PyTorch Geometric 家族
#   ⚠️ wheel 源目录名是 `torch-2.2.0+cu121`（不是 2.2.2），写错会 404：
#     pip install torch-geometric==2.5.2
#     pip install torch-scatter torch-sparse torch-cluster \
#       -f https://data.pyg.org/whl/torch-2.2.0+cu121.html
#   无 GPU 时把末尾路径换成 torch-2.2.0+cpu.html
# ============================================================
torch-geometric==2.5.2

# ============================================================
# 【第 3 步】化学与通用依赖（这批可以直接 pip 装）
#   注：官方 environment.yml 里的 rdchiral_cpp 只有 Linux 版且仅供训练，
#   推理用纯 Python 的 rdchiral 即可。
# ============================================================
rdkit==2023.9.6
rdchiral
pandas
scipy
h5py
omegaconf
antlr4-python3-runtime
tqdm
joblib
more-itertools
pebble
pyyaml

# 逆合成框架
syntheseus==0.8.0
syntheseus-graphium==0.1.0

# 网页服务（web/wheels/ 里有 Flask 的离线 wheel，断网环境可用）
flask
requests

# 【第 4 步】RetroChimera 本体（不在本仓库，需单独获取后本地安装）
#   cd <retrochimera 仓库根>
#   pip install -e .
